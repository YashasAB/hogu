Scope: Add /dating landing + protected dating app, with a complete backend and database.
0) Product Rules (Non-Negotiable)
App name: Hogu.
Auth: email + password (no OTP).
Messaging: optional Admin ↔ User single thread (no user-to-user chat).
Match states: NEW → SCHEDULE → SCHEDULED → (CANCELLED).
User actions only on Match Page:
NEW: set intent (Yes/No/Maybe).
SCHEDULE: submit availability windows.
SCHEDULED: read-only booking (venue/time).
Profile must include: city, neighborhood (free text), preferred date neighborhoods (multi from: Kalyan Nagar, HSR Layout, Whitefield, Central BLR, Indiranagar, Koramangala), Interested In (multi: MALE/FEMALE/NON_BINARY).
Admin finalizes venue/time/deposit offline, then writes booking + flips match to SCHEDULED.

5) Backend — Business Logic
5.1 Intent upsert
async function upsertIntent(matchId: bigint, userId: bigint, intent: 'YES'|'NO'|'MAYBE'|'WITHDRAWN') {
  await prisma.matchIntent.upsert({
    where: { matchId_userId: { matchId, userId } },
    update: { intent },
    create: { matchId, userId, intent }
  });
  const match = await prisma.match.findUnique({
    where: { id: matchId },
    include: { intents: true }
  });
  const i1 = match?.intents.find(i => i.userId === match?.userId1)?.intent;
  const i2 = match?.intents.find(i => i.userId === match?.userId2)?.intent;
  if (i1 === 'NO' || i2 === 'NO') {
    await prisma.match.update({ where: { id: matchId }, data: { status: 'CANCELLED' } });
  } else if (i1 === 'YES' && i2 === 'YES') {
    await prisma.match.update({ where: { id: matchId }, data: { status: 'SCHEDULE' } });
  }
}
5.2 Availability replace
async function replaceAvailability(matchId: bigint, userId: bigint, slots: Array<{date:string,start:string,end:string,tz:string}>) {
  await prisma.availabilitySlot.updateMany({
    where: { matchId, userId, status: 'ACTIVE' },
    data: { status: 'WITHDRAWN' }
  });
  await prisma.availabilitySlot.createMany({
    data: slots.map(s => ({
      matchId, userId,
      date: new Date(s.date + 'T00:00:00Z'),
      startTime: s.start,
      endTime: s.end,
      tz: s.tz,
      status: 'ACTIVE'
    }))
  });
}
5.3 Booking confirm → SCHEDULED
async function confirmBooking(matchId: bigint, payload: {
  venueId?: number; venueName?: string; venueAddress?: string;
  startTs: string; endTs: string; status: 'HELD'|'CONFIRMED'|'CANCELLED';
  paymentStatus: 'PENDING'|'PAID'|'REFUNDED';
}) {
  await prisma.booking.upsert({
    where: { matchId },
    update: { ...payload },
    create: { matchId, ...payload }
  });
  if (payload.status === 'CONFIRMED') {
    await prisma.match.update({ where: { id: matchId }, data: { status: 'SCHEDULED' } });
  }
}
5.4 Mutual seeking guard
async function isMutuallyCompatible(aId: bigint, bId: bigint) {
  const [a, b] = await Promise.all([
    prisma.user.findUnique({ where: { id: aId }, include: { seekingGenders: true } }),
    prisma.user.findUnique({ where: { id: bId }, include: { seekingGenders: true } })
  ]);
  const aSeeks = new Set(a?.seekingGenders.map(s => s.seeking));
  const bSeeks = new Set(b?.seekingGenders.map(s => s.seeking));
  const okA = aSeeks.has(b?.gender as any);
  const okB = bSeeks.has(a?.gender as any);
  return okA && okB;
}
